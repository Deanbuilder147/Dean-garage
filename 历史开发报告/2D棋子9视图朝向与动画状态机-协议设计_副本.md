### 1.3 字段默认值策略

当 unit 对象**尚未携带** `direction` / `actionState` 时（如旧存档或后端未升级），绘制层使用默认值：

```javascript
const DEFAULTS = {
  direction: 0,          // 默认正北
  actionState: 'idle',   // 默认待机
}
```

这意味着：
- **后端升级可渐进进行**：未设置时前端自动降级为默认朝向/待机姿态
- **绘制层不报错**：`??` 或 `||` 操作符合并

## 二、9 视图方向映射枚举定义

### 2.1 方向编码表

```
      NW(7)    N(0)    NE(1)
         \      |      /
          \     |     /
           \    |    /
    W(6) ———— [UNIT] ———— E(2)
           /    |    \
          /     |     \
         /      |      \
      SW(5)    S(4)    SE(3)

              TOP(8)
           (俯视视角)
```

| direction | 方向名 | 英文标识 | 六边形邻格映射 | 说明 |
|-----------|--------|----------|----------------|------|
| **0** | 正北 | N | N/A | 默认朝向，机体面朝上 |
| **1** | 东北 | NE | 尖顶 Even-R: (q, r-1) | 右上前方 |
| **2** | 东 | E | (q+1, r) | 正右 |
| **3** | 东南 | SE | (q, r+1) | 右下 |
| **4** | 南 | S | N/A | 面朝下 |
| **5** | 西南 | SW | (q-1, r+1) | 左下 |
| **6** | 西 | W | (q-1, r) | 正左 |
| **7** | 西北 | NW | (q-1, r-1) | 左上 |
| **8** | 俯视 | TOP | N/A | 正上方俯视，用于地图总览 |

### 2.2 方向枚举常量定义

```javascript
// ===== 9-View Direction Enum =====
// 位置: frontend/src/constants/unitDirections.js 或 hexUtils.js 扩展

export const DIRECTIONS = Object.freeze({
  N:  0,
  NE: 1,
  E:  2,
  SE: 3,
  S:  4,
  SW: 5,
  W:  6,
  NW: 7,
  TOP: 8,
})

export const DIRECTION_LABELS = Object.freeze({
  0: 'N',  1: 'NE', 2: 'E',  3: 'SE',
  4: 'S',  5: 'SW', 6: 'W',  7: 'NW',
  8: 'TOP',
})

/** 方向总数（不含 TOP 则为 8，含 TOP 为 9） */
export const DIRECTION_COUNT = 9

/**
 * 根据六角格坐标增量自动计算朝向
 * @param {number} fromQ - 起始列
 * @param {number} fromR - 起始行
 * @param {number} toQ - 目标列
 * @param {number} toR - 目标行
 * @returns {number} direction (0-7)，如果相同则返回当前朝向不变
 *
 * Even-R 尖顶六边形的 6 个邻格方向映射:
 *   NE: (q+1, r-1) or (q, r-1) 取决于 r 的奇偶
 *   E:  (q+1, r)
 *   SE: (q+1, r+1) or (q, r+1)
 *   SW: (q-1, r+1) or (q, r+1) // 取决于 r 的奇偶
 *   W:  (q-1, r)
 *   NW: (q-1, r-1) or (q, r-1)
 *
 * 注意: Even-R 下偶数行和奇数行的邻格偏移不同，
 * 建议使用 hexUtils.getHexNeighbors 的逆推逻辑，
 * 或直接用角度法 (atan2) 做最近方向量化。
 */
export function computeDirection(fromQ, fromR, toQ, toR) {
  if (fromQ === toQ && fromR === toR) return null // 同格不改变朝向

  // 方法: 将 (dq, dr) 映射到最近的 8 方向之一
  // 由于 hexUtils.getHexNeighbors 已提供邻格计算，
  // 这里使用角度量化法（容忍非邻格移动的退化为方向）
  const dx = toQ - fromQ
  const dy = toR - fromR

  // 将偏移转为角度（屏幕坐标系 Y 向下）
  let angle = Math.atan2(dy, dx) * (180 / Math.PI)
  if (angle < 0) angle += 360

  // 8 方向扇区量化 (每个扇区 45°)
  //   0(N):   337.5°–22.5°
  //   1(NE):  22.5°–67.5°
  //   2(E):   67.5°–112.5°
  //   3(SE):  112.5°–157.5°
  //   4(S):   157.5°–202.5°
  //   5(SW):  202.5°–247.5°
  //   6(W):   247.5°–292.5°
  //   7(NW):  292.5°–337.5°
  const octant = Math.round(angle / 45) % 8
  return octant
}
```

### 2.3 方向与六边形邻格精确映射

由于 Even-R Offset 下偶数行和奇数行的邻格偏移模式不同，**首选 `computeDirection` 角度量化法**，而非 Q/R 硬编码差分。`computeDirection` 对非邻格移动（如传送）也有合理的退化行为：落在扇区最近方向。

若需要精确映射后缀，可结合 `hexUtils.getHexNeighbors`：

```javascript
/**
 * 查找 from→to 对应的方向（0-7），必须在邻格内才返回方向，
 * 否则返回 null（不改变朝向）
 */
export function computeDirectionStrict(fromQ, fromR, toQ, toR) {
  const neighbors = getHexNeighbors(fromQ, fromR)
  const idx = neighbors.findIndex(n => n.q === toQ && n.r === toR)
  if (idx === -1) return null

  // getHexNeighbors 返回的顺序是:
  // Even-R: [NE, E, SE, SW, W, NW] (6个方向)
  // 映射到我们的 8 方向编码:
  const NEIGHBOR_TO_DIRECTION = [1, 2, 3, 5, 6, 7] // NE,E,SE,SW,W,NW
  return NEIGHBOR_TO_DIRECTION[idx]
}
```

## 三、切图资源映射字典 (Sprite Resolver)

### 3.1 设计原则

> **严禁在 Canvas 绘制函数里硬编码图片路径。**

- 资源路径/Atlas 坐标全部由配置层集中管理
- 拼合规则: `{unitCode} + {direction} + {actionState}` → `Image` 或 `{x, y, w, h}` 裁剪坐标
- 支持两种资源组织模式：
  - **Atlas 模式（推荐）**: 一张大 Sprite Sheet 包含所有机体 × 所有方向 × 所有动作
  - **独立图片模式（降级）**: 每个切片独立 PNG，URL 动态组合

### 3.2 核心接口

```javascript
// ===== Sprite Resolver =====
// 位置: frontend/src/resolvers/unitSpriteResolver.js

/**
 * 切片纹理查询结果
 * @typedef {Object} SpriteTexture
 * @property {HTMLImageElement} image   - 图片对象
 * @property {number} sx               - Atlas 源X (独立图片模式为 0)
 * @property {number} sy               - Atlas 源Y (独立图片模式为 0)
 * @property {number} sw               - 源宽度
 * @property {number} sh               - 源高度
 * @property {number} renderW          - 渲染目标宽度 (画布上大小)
 * @property {number} renderH          - 渲染目标高度
 * @property {number} anchorX          - 锚点X (相对于图片中心的偏移，底部对齐)
 * @property {number} anchorY          - 锚点Y (通常=renderH/2 即脚底在中心)
 */

/**
 * 切图解析器
 *
 * 两种工作模式，通过配置切换:
 *   1. atlas 模式: 预加载所有机体共用的 sprite sheet
 *   2. standalone 模式: 每个切片独立 URL
 */
export const unitSpriteResolver = {
  // === 配置 ===
  /** 资源模式: 'atlas' | 'standalone' */
  _mode: 'standalone',

  /** 图片缓存: key → HTMLImageElement */
  _cache: {},

  /** Atlas 元数据 (atlas 模式下使用) */
  _atlasMeta: null,

  /**
   * 初始化 (Phase 2 实现时调用)
   * @param {'atlas'|'standalone'} mode
   * @param {Object} atlasMeta - atlas 模式的元数据描述文件
   */
  init(mode, atlasMeta = null) {
    this._mode = mode
    this._atlasMeta = atlasMeta
  },

  // === 核心查询 ===

  /**
   * 获取单位纹理切片
   *
   * @param {string} unitCode     - 机体代号 (如 "KMF-001")
   * @param {number} direction    - 朝向 (0-8)
   * @param {string} actionState  - 动作状态 ('idle'|'move'|'attack'|'damaged'|'defend')
   * @param {string} fallbackUnitCode - 降级机体代号 (当前 unit 无切图时使用)
   * @returns {SpriteTexture|null} 纹理对象，未加载完成返回 null
   *
   * 查询优先级:
   *   1. unitCode_direction_actionState (精确匹配)
   *   2. unitCode_direction_idle       (该朝向降级为待机)
   *   3. unitCode_0_idle               (降级为正北待机)
   *   4. fallbackUnitCode_0_idle       (降级为通用单位默认)
   *   5. null                          (无可用资源，绘制 fallback 字母)
   */
  getTexture(unitCode, direction, actionState, fallbackUnitCode = 'DEFAULT') {
    // 查询优先级链
    const keys = [
      `${unitCode}_${direction}_${actionState}`,
      `${unitCode}_${direction}_idle`,
      `${unitCode}_0_idle`,
      `${fallbackUnitCode}_0_idle`,
    ]

    for (const key of keys) {
      const tex = this._resolve(key)
      if (tex) return tex
    }
    return null
  },

  /**
   * 内部: 根据 key 解析纹理
   * atlas 模式: 查询 atlas JSON → 裁剪坐标
   * standalone 模式: 拼接 URL → 异步加载 → 缓存
   */
  _resolve(key) {
    if (this._cache[key]) {
      const img = this._cache[key]
      if (img.complete && img.naturalWidth > 0) {
        return {
          image: img,
          sx: 0, sy: 0,
          sw: img.naturalWidth, sh: img.naturalHeight,
          renderW: img.naturalWidth, renderH: img.naturalHeight,
          anchorX: img.naturalWidth / 2,
          anchorY: img.naturalHeight,  // 脚底对齐
        }
      }
    }

    // 降级: 触发异步加载，本次返回 null
    if (this._mode === 'standalone') {
      const url = `/assets/sprites/units/${key}.png`
      if (!this._cache[key]) {
        const img = new Image()
        img.src = url
        this._cache[key] = img
      }
    }
    return null
  },

  /**
   * 预加载指定单位的所有切片（异步）
   * @param {string[]} unitCodes - 机体代码列表
   * @returns {Promise<void>}
   */
  async preload(unitCodes) {
    const directions = [0,1,2,3,4,5,6,7,8]
    const actions = ['idle','move','attack','damaged']
    const promises = []

    for (const code of unitCodes) {
      for (const d of directions) {
        for (const a of actions) {
          const key = `${code}_${d}_${a}`
          if (this._cache[key]) continue
          const img = new Image()
          img.src = `/assets/sprites/units/${key}.png`
          this._cache[key] = img
          promises.push(new Promise((resolve) => {
            img.onload = resolve
            img.onerror = resolve  // 缺失切片不阻塞
          }))
        }
      }
    }
    await Promise.all(promises)
  },
}
```

## 四、Canvas 垂直站立解耦绘制的数学实现方案

### 4.1 问题分析

当前 CTM 管线（HexGridCanvas.draw()）：

```
ctx.translate(offsetX, offsetY)     // Step 1: 平移
ctx.scale(scale, scale)              // Step 2: 缩放
ctx.transform(scX, shY, shX, scY)   // Step 3: ISO 倾斜矩阵
     ↓
drawBattleScene(ctx, ...)  // 父层在世界坐标空间绘制
```

**当前 unit 绘制**: 直接在 world-space (flatX, flatY) 绘制 → 经过 ISO 矩阵压扁 → **棋子跟着地板一起形变**。

**目标**: 棋子以 1:1 正常比例"垂直站立"在地板上，不被 ISO 矩阵压扁扭曲（Billboard 公告牌效果）。

### 4.2 数学推导

给定 unit 在六角格 `(q, r)` 的世界空间坐标:

```
flatX, flatY = pointyTopCenter(q, r, HEX_RADIUS, spacingH, spacingV)
```

该单元经过完整 CTM 后在屏幕上的位置为:

$$\begin{pmatrix} screenX \\ screenY \end{pmatrix} = \begin{pmatrix} offsetX \\ offsetY \end{pmatrix} + s \cdot \begin{pmatrix} scX & shX \\ shY & scY \end{pmatrix} \begin{pmatrix} flatX \\ flatY \end{pmatrix}$$

即：

```javascript
screenX = offsetX + scale * (ISO.scaleX * flatX + ISO.shearX * flatY)
screenY = offsetY + scale * (ISO.shearY * flatX + ISO.scaleY * flatY)
```

带入当前基准值 `(scX=1.00, scY=0.39, shX=0.25, shY=0.44)`：

```javascript
screenX = offsetX + scale * (1.00 * flatX + 0.25 * flatY)
screenY = offsetY + scale * (0.44 * flatX + 0.39 * flatY)
```

### 4.3 解耦绘制流程

```
FOR EACH unit (sorted by screenY for Z-order):
  ┌─────────────────────────────────────────────────┐
  │ Step A: 计算屏幕空间锚点                        │
  │   screenX = ox + s * (scX*flatX + shX*flatY)   │
  │   screenY = oy + s * (shY*flatX + scY*flatY)   │
  │                                                   │
  │ Step B: 临时逃逸 ISO 矩阵                       │
  │   ctx.save()                                     │
  │   ctx.setTransform(1,0,0,1,0,0)  // 重置单位矩阵 │
  │                                                   │
  │ Step C: 定位到屏幕像素 + 仅应用缩放              │
  │   ctx.translate(screenX, screenY)                │
  │   ctx.scale(scale, scale)       // 缩放但 NOT 倾斜│
  │                                                   │
  │ Step D: 绘制 2D 棋子                           │
  │   ctx.drawImage(sprite.img,                      │
  │     -anchorX, -anchorY,                         │
  │     renderW, renderH)                            │
  │                                                   │
  │ Step E: 恢复 CTM（地板纹理回归 ISO 倾斜）      │
  │   ctx.restore()                                  │
  └─────────────────────────────────────────────────┘
```

### 4.4 核心代码（Phase 2 实现时使用）

以下**不是实现代码，而是协议规范中的伪代码**，Phase 2 实现时将放入 drawBattleScene：

```javascript
// ===== 伪代码: drawBattleScene 中的 unit 渲染循环 =====

// 从 HexGridCanvas 暴露的 API 获取变换参数（供 billboard 解耦使用）
// hexGrid 是 ref() 指向的 HexGridCanvas 组件实例
const iso = hexGrid.value.ISO        // reactive: { scaleX, scaleY, shearX, shearY }
const s = hexGrid.value.scale        // ref<number>
const ox = hexGrid.value.offsetX     // ref<number>
const oy = hexGrid.value.offsetY     // ref<number>

// 1. 按 screenY 排序（实现 ISO Z-order: 后方先绘 → 前方覆盖）
const sortedUnits = [...allUnits.value]
  .filter(u => u.q !== undefined)
  .sort((a, b) => {
    const aScrY = oy + s * (iso.scaleY * a.flatY + iso.shearY * a.flatX)  // 预计算或存于 sort key
    const bScrY = oy + s * (iso.scaleY * b.flatY + iso.shearY * b.flatX)
    return aScrY - bScrY
  })

// 2. 逐个绘制 unit (billboard mode)
for (const unit of sortedUnits) {
  const { flatX, flatY } = pointyTopCenter(unit.q, unit.r, HEX_RADIUS, spacingH, spacingV)

  // Step A: 计算屏幕空间锚点 (unit 脚底中心)
  const screenX = ox + s.value * (iso.scaleX * flatX + iso.shearX * flatY)
  const screenY = oy + s.value * (iso.shearY * flatX + iso.scaleY * flatY)

  // Step B: 逃逸 ISO 矩阵
  ctx.save()

  // Step C: 重置为单位矩阵，定位到屏幕像素
  ctx.setTransform(1, 0, 0, 1, 0, 0)
  ctx.translate(screenX, screenY)
  ctx.scale(s.value, s.value)  // 只缩放，不倾斜

  // Step D: 查询切图
  const direction = unit.direction ?? 0
  const actionState = unit.actionState ?? 'idle'
  const unitCode = unit.unitCode || unit.type || 'DEFAULT'
  const sprite = unitSpriteResolver.getTexture(unitCode, direction, actionState)

  if (sprite && sprite.image.complete && sprite.image.naturalWidth > 0) {
    // ---- 2D 棋子以 1:1 正常比例绘制 ----
    ctx.drawImage(
      sprite.image,
      0, 0, sprite.sw, sprite.sh,       // 源区域（或 atlas 坐标）
      -sprite.anchorX, -sprite.anchorY,  // 目标: 脚底在锚点
      sprite.renderW, sprite.renderH      // 渲染尺寸（不受 ISO 压扁）
    )
  } else {
    // Fallback: 圆形 + 字母标记（降级）
    const r = HEX_RADIUS * 0.4
    ctx.beginPath()
    ctx.arc(0, 0, r, 0, Math.PI * 2)
    ctx.fillStyle = hexToRGBA(factionColor, 0.45)
    ctx.fill()
    ctx.strokeStyle = factionColor
    ctx.lineWidth = 2
    ctx.stroke()
    ctx.fillStyle = factionColor
    ctx.font = 'bold 14px monospace'
    ctx.textAlign = 'center'
    ctx.textBaseline = 'middle'
    ctx.fillText((unit.name || 'U')[0], 0, 0)
  }

  // HP bar（同样在 billboard 空间内绘制，保证文字/血条不变形）
  const hpPct = Math.max(0, (unit.hp || 100) / 100)
  const barW = HEX_RADIUS * 0.6
  const barH = 3
  const barY = HEX_RADIUS * 0.32  // 脚底上方
  ctx.fillStyle = 'rgba(0,0,0,0.6)'
  ctx.fillRect(-barW/2, barY, barW, barH)
  ctx.fillStyle = hpPct > 0.5 ? '#13ff43' : hpPct > 0.25 ? '#ffb000' : '#ff4d4d'
  ctx.fillRect(-barW/2, barY, barW * hpPct, barH)

  // Step E: 恢复 CTM
  ctx.restore()
}
```

### 4.5 验证公式

以基准参数 `(shearX=0.25, shearY=0.44, scaleX=1.00, scaleY=0.39)` 和 `(offsetX=400, offsetY=300, scale=1.0)` 验证一个 unit 在 `(q=3, r=2)`:

```
flatX = 64 * sqrt(3) * 3 * 1.0 + (2%2===0 ? 64*sqrt(3)/2 : 0)
      = 64*1.732*3 + 32*1.732 = 332.544 + 55.424 = 387.968
flatY = 64 * 1.5 * 2 * 1.0 = 192

screenX = 400 + 1.0 * (1.00*387.968 + 0.25*192) = 400 + 387.968 + 48 = 835.968
screenY = 300 + 1.0 * (0.44*387.968 + 0.39*192) = 300 + 170.706 + 74.88 = 545.586

→ 棋子中心出现在屏幕 (836, 546) 处
→ 以 setTransform(1,0,0,1,0,0) + translate(836,546) + scale(1,1) 绘制
→ 棋子正常比例，不跟随ISO矩阵压扁 ✓
```
