### PATCH 2: 注入 glossaryConfig 响应式状态 + 核心函数
```javascript
const glossaryConfig = ref(null)

async function loadGlossaryConfig() {
  const res = await glossaryAPI.getConfig()
  glossaryConfig.value = res.data
}

// 被动技能描述动态生成（8大核心词条）
function getPassiveSkillDesc(skill) {
  const gc = glossaryConfig.value
  if (!gc?.skills) return skill.description || ''
  const gs = gc.skills[skill.type]; if (!gs) return skill.description || ''
  switch (skill.type) {
    case 'block':        return `被动：受到敌人攻击时伤害-${gs.reduction}`
    case 'execute':      return `近战伤害结算后，目标HP<${gs.hp_threshold_percent}%最大HP时直接斩杀`
    case 'focused_fire': return `放弃移动，获得固定伤害加成+${gs.bonus}`
    case 'throw':        return `主动：1~${gs.max_range}格，目标周围${gs.aoe_range}格所有目标下次伤害+${gs.value}`
    case 'sweep':        return `主动：扇形${gs.max_range}格范围攻击…`
    case 'duel':         return `双方在攻击范围内且HP<对方${gs.stat_comparison}时触发，攻击力高者胜`
    case 'snatch':       return `伤害值>被攻击者武器攻击值时触发，伤害减为×${gs.damage_multiplier}并获得武器`
    case 'lucky':        return `获得空投时可再次移动并攻击`
    case 'reactivate':   return `击杀敌军时触发，额外一回合（不连续触发）`
    default:             return skill.description || ''
  }
}

// 主动技能 tooltip 动态生成
function getActiveSkillTooltip(skill) {
  const gc = glossaryConfig.value
  if (!gc?.skills) return skill.description || ''
  const gs = gc.skills[skill.type]; if (!gs) return skill.description || ''
  switch (skill.type) {
    case 'focused_fire': return `${skill.name}: 放弃移动，固定伤害加成 +${gs.bonus}`
    case 'sweep':        return `${skill.name}: 扇形${gs.max_range}格范围，精准伤害${gs.damage_modifier_precise}`
    case 'throw':        return `${skill.name}: 1~${gs.max_range}格，AOE伤害+${gs.value}`
    default:             return skill.description || skill.name || ''
  }
}
```

## 六、技术要点

1. **零硬编码**: 所有 8 大核心技能的数字描述全部替换为 `glossaryConfig.skills.{type}.{field}` 模板字符串
2. **响应式刷新**: `ref(null)` + `loadGlossaryConfig()` 在 onMounted 和 refreshState 两处调用，确保页面切换后数据最新
3. **容错降级**: 当 glossaryConfig 未加载时，回退到 `skill.description` 原始字段，保证 UI 不崩溃
4. **ESM/CJS 桥接**: `createRequire` 解决 module 系统混用问题
5. **热加载**: configLoader 的 `clearCache()` 机制确保 POST 保存后 GET 立即返回最新值
